    <script>
        const yes = document.getElementById("yes");
        const no = document.getElementById("no");
        const warning = document.getElementById("warning");
        const bgSong = document.getElementById("bgSong");
        const boomSound = document.getElementById("boomSound");
        const yesSong = document.getElementById("yesSong");
        let attempts = 0;
        let musicStarted = false;

        function startBgMusic() {
            if (!musicStarted) {
                bgSong.volume = 0.5;
                bgSong.play().catch(err => console.log("Audio play blocked by browser."));
                musicStarted = true;
            }
        }

        // NO BUTTON RUNS AWAY
        function escapeButton() {
            attempts++;
            startBgMusic();

            // After several attempts
            if (attempts >= 3) {
                warning.style.display = "block";
                no.style.display = "none";
                document.querySelector(".subtitle").style.display = "none";
                // Play dramatic boom sound when the gun cat appears
                boomSound.play();
                return;
            }

            const maxX = window.innerWidth - no.offsetWidth - 20;
            const maxY = window.innerHeight - no.offsetHeight - 20;
            const x = Math.max(10, Math.random() * maxX);
            const y = Math.max(10, Math.random() * maxY);

            no.style.position = "fixed";
            no.style.left = x + "10px";
            no.style.top = y + "5px";
        }

        // Mouse events for PC
        no.addEventListener("mouseover", escapeButton);
        
        // Touch events for Mobile
        no.addEventListener("touchstart", function(e) {
            e.preventDefault();
            escapeButton();
        });

        // If somehow clicked
        no.addEventListener("click", escapeButton);

        // YES BUTTON
        yes.addEventListener("click", function() {
            document.getElementById("main").style.display = "none";
            document.getElementById("final").style.display = "block";

            // Switch songs
            bgSong.pause();
            yesSong.currentTime = 0;
            yesSong.play().catch(function(error) {
                console.log("Music needs user interaction.");
            });

            // Hearts effect
            createHearts();
        });

        // FLOATING HEARTS
        function createHearts() {
            setInterval(function() {
                const heart = document.createElement("div");
                heart.className = "heart";
                heart.innerHTML = ["❤️", "💖", "💝", "💕", "🌹"][Math.floor(Math.random() * 5)];
                heart.style.left = Math.random() * 100 + "vw";
                heart.style.fontSize = (15 + Math.random() * 25) + "px";
                heart.style.duration = (3 + Math.random() * 3) + "s";
                document.body.appendChild(heart);
                
                setTimeout(function() {
                    heart.remove();
                }, 5000);
            }, 300);
        }
    </script>
